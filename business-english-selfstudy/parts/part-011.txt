  if(idx>=diagnosticQuestions.length){

    var rec=score<=4?'Module 1 → Module 2 → Toolkit 순서로 기초 구조와 표현을 먼저 확인하세요.':
            score<=8?'Module 2 Frameworks와 Toolkit을 먼저 복습한 뒤 Scenarios·Interview로 이동하세요.':

      '<div class="lab-card" style="text-align:left"><b>권장 시작점</b><p class="lab-intro" style="margin-top:6px">'+rec+'</p></div>'+
