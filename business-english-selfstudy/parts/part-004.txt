function qlist(items, compact){
  return '<ul class="qlist'+(compact?' compact':'')+'">'+items.map(function(t){return '<li>'+t+'</li>';}).join('')+'</ul>';
}
function steplist(items){
  return '<ol class="steplist">'+items.map(function(t){return '<li>'+t+'</li>';}).join('')+'</ol>';
}
function chips(items){
  return '<div class="word-chip-wrap">'+items.map(function(t){return '<span class="word-chip">'+t+'</span>';}).join('')+'</div>';
}
function head(en, kr){
  return '<div class="slide-eyebrow">'+en+'</div><h2 class="slide-title">'+ (kr && !en.match(/^Toolkit/) ? '' : '') + '</h2>';
}
function slideHead(title, kr){
  return '<h2 class="slide-title">'+title+'</h2>'+(kr?'<p class="slide-kr">'+kr+'</p>':'');
}
function block(title, tagKind){
  var tag = tagKind ? '<span class="tag '+tagKind+'">'+(tagKind==='kr'?'KR':'EN')+'</span>' : '';
  return '<h3 class="blk">'+title+' '+tag+'</h3>';
}
function example(html){ return '<div class="example">'+html+'</div>'; }
function note(html){ return '<div class="note-box">'+html+'</div>'; }

    return '<div class="star-step"><div class="star-letter">'+s.l+'</div><div class="star-body"><b>'+s.t+'</b><span>'+s.d+'</span></div></div>';
  }).join('');
}

}
