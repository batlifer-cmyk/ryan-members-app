function bindStudyInputs(root){
  (root||document).querySelectorAll('[data-study-key]').forEach(function(el){
    var key=el.getAttribute('data-study-key');
    if(el.type==='checkbox') el.checked=!!studyState.answers[key];
    else if(getAnswer(key) && !el.value) el.value=getAnswer(key);
    el.addEventListener('input',function(){

    });
  });
}

function dictateTo(key,btn){
  var SR=window.SpeechRecognition||window.webkitSpeechRecognition;
  if(!SR){ alert('음성 인식은 지원 브라우저(예: Chrome 계열)에서 사용할 수 있습니다.'); return; }
  var r=new SR(); r.lang='en-US'; r.interimResults=false; btn.disabled=true; var old=btn.textContent; btn.textContent='Listening...';
  r.onresult=function(e){
    var t=e.results[0][0].transcript;
    var el=document.querySelector('[data-study-key="'+CSS.escape(key)+'"]');
    if(el){ el.value=(el.value?el.value+' ':'')+t; setAnswer(key,el.value); }
  };
  r.onend=function(){btn.disabled=false;btn.textContent=old;};
  r.start();
}
var activeTimer=null;
function startStudyTimer(seconds,face,meter){
  if(activeTimer) clearInterval(activeTimer);
  var left=seconds, total=seconds;
  function tick(){
    face.textContent=Math.floor(left/60)+':'+String(left%60).padStart(2,'0');
    if(meter) meter.style.width=(100*(1-left/total))+'%';
    if(left<=0){ clearInterval(activeTimer); activeTimer=null; face.textContent='TIME'; return; }
    left--;
  }
  tick(); activeTimer=setInterval(tick,1000);
}
