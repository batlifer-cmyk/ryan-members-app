function centerCard(key,title,desc){
  var p=studyState.training.byPack[key]||{sessions:0,correct:0,attempts:0};

  return '<div class="practice-center-card"><h4>'+title+'</h4><p>'+desc+'</p>'+

function getAllTasks(){
  var out=[];Object.keys(TRAINING_PACKS).forEach(function(k){TRAINING_PACKS[k].forEach(function(t){var c=Object.assign({},t);c.pack=k;out.push(c);});});return out;
}
function startWeakReview(){
  var all=getAllTasks(), ids=Object.keys(studyState.training.weak||{}).filter(function(id){return studyState.training.weak[id]>0;});
  var tasks=all.filter(function(t){return ids.indexOf(t.id)!==-1;});
  if(!tasks.length){alert('현재 저장된 오답이 없습니다.');return;}
  startTraining('weak',Math.min(10,tasks.length),tasks);
}
var trainingSession=null;
function startTraining(pack,count,forcedTasks){
  ensureTrainerUI();
  var source;
  if(forcedTasks){ source=forcedTasks.slice(); }
  else if(pack==='mixed'){source=getAllTasks();}
  else {source=(TRAINING_PACKS[pack]||[]).map(function(t){var c=Object.assign({},t);c.pack=pack;return c;});}
  if(!source.length){alert('이 모듈에는 아직 훈련 문제가 없습니다.');return;}
  var weak=studyState.training.weak||{};
  var priority=source.filter(function(t){return weak[t.id]>0;}).sort(function(a,b){return (weak[b.id]||0)-(weak[a.id]||0);});
  var used={};priority.forEach(function(t){used[t.id]=true;});
  var rest=shuffled(source.filter(function(t){return !used[t.id];}));
  var chosen=priority.concat(rest).slice(0,Math.min(count,source.length));
  trainingSession={pack:pack,queue:chosen,index:0,correct:0,graded:0,productive:0,retries:0,firstPass:chosen.length};
  document.getElementById('trainer-overlay').classList.add('open');
  document.body.style.overflow='hidden';
  renderTrainingTask();
}
function closeTraining(){
  var o=document.getElementById('trainer-overlay'); if(o)o.classList.remove('open');
  document.body.style.overflow='';
  if(document.getElementById('practice-center-live')) refreshPracticeCenter();
}
function taskPack(t){
  if(t.pack)return t.pack;
  for(var k in TRAINING_PACKS) if(TRAINING_PACKS[k].some(function(x){return x.id===t.id;})) return k;
  return trainingSession && trainingSession.pack!=='mixed'?trainingSession.pack:'mixed';
}
