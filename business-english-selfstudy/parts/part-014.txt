  var root=btn.closest('#content-inner'),rows=Array.from(root.querySelectorAll('.idiom-row')).map(function(r){return r.textContent.trim();});var el=btn.parentElement.querySelector('#idiom-random');el.textContent=randomOf(rows);

function newCollocationQuiz(btn){
  var root=btn.closest('#content-inner'),rows=Array.from(root.querySelectorAll('table.voc-table tr')).map(function(r){var t=r.querySelectorAll('td');return t.length>=3?[t[0].textContent.trim(),t[2].textContent.trim()]:null;}).filter(Boolean);
  if(rows.length<4)return;var target=randomOf(rows),verbs=rows.map(function(r){return r[0];}).filter(function(v){return v!==target[0];});
  var opts=[target[0]].concat(verbs.sort(function(){return Math.random()-.5;}).slice(0,3)).sort(function(){return Math.random()-.5;});
  var slot=btn.parentElement.querySelector('#collocation-quiz');
  slot.innerHTML='<div class="lab-question">Which verb completes: ___ '+escHtml(target[1])+'?</div>'+opts.map(function(v){return '<button class="quiz-option" data-v="'+escHtml(v)+'">'+escHtml(v)+'</button>';}).join('');
  slot.querySelectorAll('.quiz-option').forEach(function(b){b.onclick=function(){var ok=b.dataset.v===target[0];b.classList.add(ok?'correct':'wrong');slot.querySelectorAll('.quiz-option').forEach(function(z){z.disabled=true;});if(!ok)Array.from(slot.querySelectorAll('.quiz-option')).find(function(z){return z.dataset.v===target[0];}).classList.add('correct');};});
}

function mountPractice(item){
  var root=document.getElementById('content-inner'); if(!root) return;
  if(item.sec.id==='0-0'){renderDiagnostic();return;}
  if(moduleQuiz[item.sec.id]) root.insertAdjacentHTML('beforeend',microQuizHtml(item.sec.id));
  var h='';
  if(item.mod.id==='m0') h=overviewPractice(item,root);
  if(item.mod.id==='m1' && item.sec.id!=='1-0') h=smallTalkPractice(item,root);
  if(item.mod.id==='m2' && item.sec.id!=='2-0') h=frameworkPractice(item,root);
  if(item.mod.id==='m3') h=scenarioPractice(item,root);
  if(item.mod.id==='m4') h=tedPractice(item,root);
  if(item.mod.id==='m5') h=interviewPractice(item,root);
  if(item.mod.id==='m6') h=projectPractice(item,root);
  if(item.mod.id==='tk') h=toolkitPractice(item,root);
  if(h) root.insertAdjacentHTML('beforeend',h);
  bindStudyInputs(root);
  if(item.sec.id==='d-1'){var b=root.querySelector('#practice-d-1 .lab-btn'); if(b)newCollocationQuiz(b);}
  updateStudyCount();
}
